class Solution {
public:
    int largestPerimeter(vector<int>& nums) {//to be a valid triange the a+b>c (sorted acsending)
    int currentLargest=0;
    std::sort(nums.begin(),nums.end());
    for(int i = nums.size()-1; i>1;i--)
    {
        if(nums[i-1]+nums[i-2]>nums[i]&&nums[i-1]+nums[i-2]+nums[i]>currentLargest)currentLargest=nums[i-1]+nums[i-2]+nums[i];
    }

        return currentLargest;
    }
};
