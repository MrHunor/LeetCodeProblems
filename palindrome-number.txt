#include <iostream>
using namespace std;
class Solution {
public:
    int getN(int x, int n )
    {
     for( int i = 0; i<n; i++)
     {
        x=x/10;
     }
     return x % 10;
    }
    bool isPalindrome(int x) {
        if(x<0)return false;
        int length= 0; 
        int x_ = x;
        for(length = 0; x_!=0; length++ )
        {
            x_ = x_/10;
        }
    for (int i = 0; i<length; i++)
    {
    if(getN(x,i)!=getN(x,length-1-i))return false;
    }
    return true;
    }
};
