class Solution {
public:
    void setZeroes(vector<vector<int>>& matrix) {
std::vector<int> markedRows;
std::vector<int> markedCollums;
     
        for(int y = 0; y<matrix.size(); y++)
        {
            for(int x = 0; x<matrix[y].size(); x++)
            {
                if(matrix[y][x]==0)
                {
               markedRows.push_back(y);
               markedCollums.push_back(x);   

                }
            }
        }
                   for(int i = 0; i <markedRows.size();i++)
                   {

                    for(int x_ = 0; x_ < matrix[markedRows[i]].size(); x_++)
                    {
                        matrix[markedRows[i]][x_]=0;
                    }
                   }
                   for(int i = 0; i < markedCollums.size(); i++)
                   {
                    for(int y_ = 0; y_< matrix.size();y_++)
                    {
                        matrix[y_][markedCollums[i]]=0;
                    }
                   }
                    
    }
};
